48 @Override
49 public void handle(HttpExchange exchange) throws IOException {
50 exchange.getResponseHeaders().set("Access-Control-Allow-Origin", "*");
51 exchange.getResponseHeaders().set("Access-Control-Allow-Methods", "GET, POST, OPTIONS");
52 exchange.getResponseHeaders().set("Access-Control-Allow-Headers", "Content-Type");
53
54 if ("OPTIONS".equalsIgnoreCase(exchange.getRequestMethod())) {
55 exchange.sendResponseHeaders(204, -1);
56 return;
57 }
58
59 String path = exchange.getRequestURI().getPath();
60
61 try {
62 if ("/api/upload".equals(path) && "POST".equalsIgnoreCase(exchange.getRequestMethod())) {
63 handleUpload(exchange);
64 } else if ("/api/match".equals(path) && "POST".equalsIgnoreCase(exchange.getRequestMethod())) {
65 handleMatch(exchange);
66 } else if ("/api/status".equals(path) && "GET".equalsIgnoreCase(exchange.getRequestMethod())) {
67 handleStatus(exchange);
68 } else if ("/api/samples".equals(path) && "GET".equalsIgnoreCase(exchange.getRequestMethod())) {
69 handleListSamples(exchange);
70 } else if ("/api/oct22-data".equals(path) && "GET".equalsIgnoreCase(exchange.getRequestMethod())) {
71 handleLoadOct22Data(exchange);
72 } else if ("/api/export/oct22-master".equals(path) && "GET".equalsIgnoreCase(exchange.getRequestMethod())) {
73 handleExportOct22Master(exchange);
74 } else if ("/api/export/master".equals(path) && "POST".equalsIgnoreCase(exchange.getRequestMethod())) {
75 handleExportMaster(exchange);
76 } else if ("/api/export/sensor".equals(path) && "POST".equalsIgnoreCase(exchange.getRequestMethod())) {
77 handleExportSensor(exchange);
78 } else if (path.startsWith("/api/sample-data/") && "GET".equalsIgnoreCase(exchange.getRequestMethod())) {
79 handleGetSampleData(exchange, path.substring("/api/sample-data/".length()));
80 } else {
81 sendJson(exchange, 404, Map.of("error", "Endpoint not found: " + path));
82 }
83 } catch (Exception e) {
84 e.printStackTrace();

89 private void handleUpload(HttpExchange exchange) throws IOException {
90 String body = new String(exchange.getRequestBody().readAllBytes(), StandardCharsets.UTF_8);
91 JsonNode root = mapper.readTree(body);
92
93 if (root == null || !root.has("sensorType") || !root.has("csvContent")) {
94 sendJson(exchange, 400, Map.of("error", "Missing required fields: sensorType and csvContent."));
95 return;
96 }
97
98 String sensorTypeId = root.get("sensorType").asText();
99 String filename = root.has("filename") ? root.get("filename").asText() : "uploaded.csv";
100
101 String csvContent = "";
102 JsonNode contentNode = root.get("csvContent");
103 if (contentNode.isTextual()) {
104 csvContent = contentNode.asText();
105 } else if (contentNode.isBinary()) {
106 csvContent = new String(contentNode.binaryValue(), StandardCharsets.UTF_8);
107 }
108
109 SensorType sensorType = SensorType.fromId(sensorTypeId);
110 if (sensorType == null) {
111 if ("vehicle_dynamics".equalsIgnoreCase(sensorTypeId) || "vehicle_motion".equalsIgnoreCase(sensorTypeId)) {
112 SensorParseResult vel = CsvSensorParser.parse(csvContent, SensorType.VEHICLE_VELOCITY, filename);
113 SensorParseResult acc = CsvSensorParser.parse(csvContent, SensorType.VEHICLE_ACCELERATION, filename);
114 Map<String, Object> combo = new LinkedHashMap<>();
115 combo.put("success", vel.isSuccess() || acc.isSuccess());
116 combo.put("velocity", vel);
117 combo.put("acceleration", acc);
118 sendJson(exchange, 200, combo);
119 return;
120 }
121 sendJson(exchange, 400, Map.of("error", "Unknown sensor type: " + sensorTypeId));
122 return;
123 }
124
125 SensorParseResult result = CsvSensorParser.parse(csvContent, sensorType, filename);
126 sendJson(exchange, 200, result);
127 }

129 private void handleMatch(HttpExchange exchange) throws IOException {
130 String body = new String(exchange.getRequestBody().readAllBytes(), StandardCharsets.UTF_8);
131 JsonNode root = mapper.readTree(body);
132
133 if (root == null || !root.has("seriesX") || !root.has("seriesY")) {
134 sendJson(exchange, 400, Map.of("error", "Missing required seriesX or seriesY objects in request body."));
135 return;
136 }
137
138 SensorParseResult seriesX = mapper.treeToValue(root.get("seriesX"), SensorParseResult.class);
139 SensorParseResult seriesY = mapper.treeToValue(root.get("seriesY"), SensorParseResult.class);
140 String labelX = root.has("labelX") ? root.get("labelX").asText() : "Series X";
141 String labelY = root.has("labelY") ? root.get("labelY").asText() : "Series Y";
142
143 MatchResult matchResult = ExactTimestampMatcher.match(seriesX, seriesY, labelX, labelY);
144 sendJson(exchange, 200, matchResult);
145 }

335 private String guessSensorType(String filename) {
336 String lower = filename.toLowerCase();
337 if (lower.contains("smartwatch") || lower.contains("samsung")) return SensorType.SMARTWATCH_HEART_RATE.getId();
338 if (lower.contains("chest") || lower.contains("ecg") || lower.contains("polar")) return SensorType.CHEST_ECG_HEART_RATE.getId();
339 if (lower.contains("dynamics") || lower.contains("motion")) return "vehicle_dynamics";
340 if (lower.contains("velocity") || lower.contains("speed")) return SensorType.VEHICLE_VELOCITY.getId();
341 if (lower.contains("accel")) return SensorType.VEHICLE_ACCELERATION.getId();
342 if (lower.contains("gsr") || lower.contains("healthypi") || lower.contains("hs_record")) return SensorType.HEALTHYPI_GSR.getId();
343 return SensorType.SMARTWATCH_HEART_RATE.getId();
344 }
