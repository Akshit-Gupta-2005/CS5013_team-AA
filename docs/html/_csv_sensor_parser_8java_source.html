40 public static SensorParseResult parse(String csvContent, SensorType sensorType, String filename) {
41 if (csvContent == null || csvContent.trim().isEmpty()) {
42 return SensorParseResult.failure(
43 sensorType != null ? sensorType.getId() : "unknown",
44 filename,
45 List.of("The uploaded CSV file is empty.")
46 );
47 }
48
49 // Clean UTF-8 BOM if present
50 String content = csvContent;
51 if (content.startsWith("\uFEFF")) {
52 content = content.substring(1);
53 }
54
55 try (StringReader reader = new StringReader(content)) {
56 CSVFormat format = CSVFormat.DEFAULT.builder()
57 .setHeader()
58 .setSkipHeaderRecord(true)
59 .setIgnoreHeaderCase(true)
60 .setTrim(true)
61 .setIgnoreEmptyLines(true)
62 .build();
63
64 CSVParser parser = format.parse(reader);
65 Map<String, Integer> headerMap = parser.getHeaderMap();
66
67 if (headerMap == null || headerMap.isEmpty()) {
68 return SensorParseResult.failure(
69 sensorType.getId(),
70 filename,
71 List.of("No CSV header row detected. The file must include column headers.")
72 );
73 }
74
75 // 1. Check if this file is a raw GNSS Logger stream (e.g. Nissan Magnite or Oct 22 Polar+GNSS log)
76 if (isGnssLog(headerMap)) {
77 return parseGnssLog(parser, headerMap, sensorType, filename);
78 }
79
80 // 2. Check if this file is a Samsung Watch (HR and IBI) multi-session export
81 if (isSamsungWatchLog(headerMap)) {
82 return parseSamsungWatchLog(parser, headerMap, sensorType, filename);
83 }
84
85 // 3. Check if this file is a HealthyPi Watch (GSR) 32-Hz recording
86 if (isHealthyPiGsrLog(headerMap)) {
87 return parseHealthyPiGsrLog(parser, headerMap, sensorType, filename);
88 }
89
90 // 4. Standard generic CSV parsing
91 String timestampCol = findColumn(headerMap, sensorType.getTimestampColumnAliases());
92 String valueCol = findColumn(headerMap, sensorType.getValueColumnAliases());
93
94 List<String> errors = new ArrayList<>();
95 if (timestampCol == null) {
96 errors.add("Missing required timestamp column. Expected one of: " +
97 String.join(", ", sensorType.getTimestampColumnAliases()) +
98 ". Found headers: " + headerMap.keySet());
99 }
100
101 if (valueCol == null) {
102 errors.add("Missing required measurement column for " + sensorType.getDisplayName() +
103 " (" + sensorType.getUnit() + "). Expected one of: " +
104 String.join(", ", sensorType.getValueColumnAliases()) +
105 ". Found headers: " + headerMap.keySet());
106 }
107
108 if (!errors.isEmpty()) {
109 return SensorParseResult.failure(sensorType.getId(), filename, errors);
110 }
111
112 List<SensorDataPoint> points = new ArrayList<>();
113 List<String> warnings = new ArrayList<>();
114 int totalRows = 0;
115 double minValue = Double.MAX_VALUE;
116 double maxValue = -Double.MAX_VALUE;
117 long minEpoch = Long.MAX_VALUE;
118 long maxEpoch = Long.MIN_VALUE;
119
120 for (CSVRecord record : parser) {
121 totalRows++;
122 String rawTs = record.get(timestampCol);
123 String rawVal = record.get(valueCol);
124
125 if (rawTs == null || rawTs.trim().isEmpty() || rawVal == null || rawVal.trim().isEmpty()) {
126 if (warnings.size() < 10) {
127 warnings.add("Row " + (totalRows + 1) + ": skipped due to empty timestamp or value.");
128 }
129 continue;
130 }
131
132 ParsedTimestamp parsedTime = parseTimestamp(rawTs.trim());
133 if (parsedTime == null) {
134 if (warnings.size() < 10) {
135 warnings.add("Row " + (totalRows + 1) + ": unable to parse timestamp '" + rawTs + "'.");
136 }
137 continue;
138 }
139
140 double val;
141 try {
142 val = Double.parseDouble(rawVal.trim());
143 if (Double.isNaN(val) || Double.isInfinite(val)) {
144 continue;
145 }
146 } catch (NumberFormatException e) {
147 if (warnings.size() < 10) {
148 warnings.add("Row " + (totalRows + 1) + ": invalid numeric value '" + rawVal + "'.");
149 }
150 continue;
151 }
152
153 points.add(new SensorDataPoint(rawTs, parsedTime.epochMillis, parsedTime.formatted, val));
154
155 if (val < minValue) minValue = val;
156 if (val > maxValue) maxValue = val;
157 if (parsedTime.epochMillis < minEpoch) minEpoch = parsedTime.epochMillis;
158 if (parsedTime.epochMillis > maxEpoch) maxEpoch = parsedTime.epochMillis;
159 }
160
161 if (points.isEmpty()) {
162 return SensorParseResult.failure(
163 sensorType.getId(),
164 filename,
165 List.of("No valid data rows could be parsed from the file.")
166 );
167 }
168
169 points.sort(Comparator.comparingLong(SensorDataPoint::getEpochMillis));
170
171 String minFormatted = DISPLAY_FORMATTER.format(Instant.ofEpochMilli(minEpoch));
172 String maxFormatted = DISPLAY_FORMATTER.format(Instant.ofEpochMilli(maxEpoch));
173
174 return new SensorParseResult(
175 true,
176 sensorType.getId(),
177 sensorType.getDisplayName(),
178 sensorType.getUnit(),
179 filename,
180 totalRows,
181 points.size(),
182 minValue,
183 maxValue,
184 minEpoch,
185 maxEpoch,
186 minFormatted,
187 maxFormatted,
188 points,
189 List.of(),
190 warnings
191 );
192
193 } catch (Exception e) {
194 return SensorParseResult.failure(
195 sensorType != null ? sensorType.getId() : "unknown",
196 filename,
197 List.of("Failed to parse CSV file: " + e.getMessage())
198 );
199 }
200 }

202 private static String findColumn(Map<String, Integer> headerMap, List<String> aliases) {
203 for (String header : headerMap.keySet()) {
204 String norm = normalize(header);
205 for (String alias : aliases) {
206 if (norm.equals(normalize(alias))) {
207 return header;
208 }
209 }
210 }
211 for (String header : headerMap.keySet()) {
212 String norm = normalize(header);
213 for (String alias : aliases) {
214 if (norm.contains(normalize(alias))) {
215 return header;
216 }
217 }
218 }
219 return null;
220 }

225 private static boolean isGnssLog(Map<String, Integer> headerMap) {
226 if (headerMap == null) return false;
227 boolean hasDateOrTime = false;
228 boolean hasSpeedOrAccel = false;
229 for (String key : headerMap.keySet()) {
230 String norm = normalize(key);
231 if (norm.equals("phonenumber") || norm.equals("measconstellationtype") || norm.equals("timeist")) {
232 return true;
233 }
234 if (norm.equals("date") || norm.equals("time") || norm.equals("timeist")) {
235 hasDateOrTime = true;
236 }
237 if (norm.equals("speedkph") || norm.equals("speedmps") || norm.equals("accelerationmps2")) {
238 hasSpeedOrAccel = true;
239 }
240 }
241 return hasDateOrTime && hasSpeedOrAccel;
242 }

244 private static SensorParseResult parseGnssLog(
245 CSVParser parser,
246 Map<String, Integer> headerMap,
247 SensorType sensorType,
248 String filename) {
249
250 String polarHrCol = findExactColumn(headerMap, "polarhrbpm");
251 boolean hasPolarHr = polarHrCol != null;
252 boolean hasTimeIst = findExactColumn(headerMap, "timeist") != null;
253
254 if (sensorType == SensorType.CHEST_ECG_HEART_RATE && !hasPolarHr) {
255 return SensorParseResult.failure(
256 sensorType.getId(),
257 filename,
258 List.of("The uploaded file is a vehicle GNSS motion log without Polar Chest ECG columns. Please upload a GNSS log containing Polar_HR_bpm or a dedicated Chest ECG CSV.")
259 );
260 }
261
262 if (sensorType != SensorType.VEHICLE_VELOCITY
263 && sensorType != SensorType.VEHICLE_ACCELERATION
264 && sensorType != SensorType.CHEST_ECG_HEART_RATE) {
265 return SensorParseResult.failure(
266 sensorType.getId(),
267 filename,
268 List.of("The uploaded file is a vehicle GNSS motion log (velocity, acceleration & Polar chest HR), but the selected target is " +
269 sensorType.getDisplayName() + ". Please upload to Vehicle Velocity, Vehicle Acceleration, or Chest ECG Monitor.")
270 );
271 }
272
273 List<SensorDataPoint> points = new ArrayList<>();
274 List<String> warnings = new ArrayList<>();
275 int totalRows = 0;
276 double minValue = Double.MAX_VALUE;
277 double maxValue = -Double.MAX_VALUE;
278 long minEpoch = Long.MAX_VALUE;
279 long maxEpoch = Long.MIN_VALUE;
280
281 int dateIdx = -1;
282 int timeIdx = -1;
283 int spdKphIdx = -1;
284 int accelIdx = -1;
285 int polarHrIdx = hasPolarHr ? headerMap.get(polarHrCol) : -1;
286 boolean detected = false;
287
288 for (CSVRecord record : parser) {
289 totalRows++;
290 int size = record.size();
291 if (size < 10) continue;
292
293 if (!detected) {
294 for (int i = 0; i < Math.min(size, 8); i++) {
295 String s = record.get(i).trim();
296 if (s.matches("^\\d{4}-\\d{2}-\\d{2}$")) {
297 dateIdx = i;
298 break;
299 }
300 }
301 if (dateIdx != -1 && dateIdx + 1 < size) {
302 timeIdx = dateIdx + 1;
303 spdKphIdx = dateIdx + 6;
304 accelIdx = dateIdx + 7;
305 if (hasPolarHr && dateIdx > 1) {
306 polarHrIdx = headerMap.get(polarHrCol) + (dateIdx - 1);
307 }
308 detected = true;
309 } else {
310 dateIdx = 2;
311 timeIdx = 3;
312 spdKphIdx = 8;
313 accelIdx = 9;
314 detected = true;
315 }
316 }
317
318 int valIdx;
319 if (sensorType == SensorType.VEHICLE_VELOCITY) {
320 valIdx = spdKphIdx;
321 } else if (sensorType == SensorType.VEHICLE_ACCELERATION) {
322 valIdx = accelIdx;
323 } else {
324 valIdx = polarHrIdx;
325 }
326
327 if (valIdx < 0 || valIdx >= size || timeIdx >= size || dateIdx >= size) {
328 continue;
329 }
330
331 String dateStr = record.get(dateIdx).trim();
332 String timeStr = record.get(timeIdx).trim();
333 String rawVal = record.get(valIdx).trim();
334
335 if (dateStr.isEmpty() || timeStr.isEmpty() || rawVal.isEmpty()) {
336 continue;
337 }
338
339 String fullTimestamp = dateStr + " " + timeStr;
340 ParsedTimestamp parsedTime = hasTimeIst
341 ? parseTimestampWithZone(fullTimestamp, IST_OFFSET)
342 : parseTimestamp(fullTimestamp);
343 if (parsedTime == null) {
344 if (warnings.size() < 10) {
345 warnings.add("Row " + (totalRows + 1) + ": unable to parse timestamp '" + fullTimestamp + "'.");
346 }
347 continue;
348 }
349
350 double val;
351 try {
352 val = Double.parseDouble(rawVal);
353 if (Double.isNaN(val) || Double.isInfinite(val)) {
354 continue;
355 }
356 if (sensorType == SensorType.CHEST_ECG_HEART_RATE && val <= 0) {
357 continue;
358 }
359 } catch (NumberFormatException e) {
360 if (warnings.size() < 10) {
361 warnings.add("Row " + (totalRows + 1) + ": invalid numeric value '" + rawVal + "'.");
362 }
363 continue;
364 }
365
366 points.add(new SensorDataPoint(fullTimestamp, parsedTime.epochMillis, parsedTime.formatted, val));
367
368 if (val < minValue) minValue = val;
369 if (val > maxValue) maxValue = val;
370 if (parsedTime.epochMillis < minEpoch) minEpoch = parsedTime.epochMillis;
371 if (parsedTime.epochMillis > maxEpoch) maxEpoch = parsedTime.epochMillis;
372 }
373
374 if (points.isEmpty()) {
375 return SensorParseResult.failure(
376 sensorType.getId(),
377 filename,
378 List.of("No valid " + sensorType.getDisplayName() + " data rows could be parsed from the GNSS log.")
379 );
380 }
381
382 points.sort(Comparator.comparingLong(SensorDataPoint::getEpochMillis));
383
384 String minFormatted = DISPLAY_FORMATTER.format(Instant.ofEpochMilli(minEpoch));
385 String maxFormatted = DISPLAY_FORMATTER.format(Instant.ofEpochMilli(maxEpoch));
386
387 return new SensorParseResult(
388 true,
389 sensorType.getId(),
390 sensorType.getDisplayName(),
391 sensorType.getUnit(),
392 filename,
393 totalRows,
394 points.size(),
395 minValue,
396 maxValue,
397 minEpoch,
398 maxEpoch,
399 minFormatted,
400 maxFormatted,
401 points,
402 List.of(),
403 warnings
404 );
405 }

416 private static SensorParseResult parseSamsungWatchLog(
417 CSVParser parser,
418 Map<String, Integer> headerMap,
419 SensorType sensorType,
420 String filename) {
421
422 if (sensorType != SensorType.SMARTWATCH_HEART_RATE) {
423 return SensorParseResult.failure(
424 sensorType.getId(),
425 filename,
426 List.of("The uploaded file is a Samsung Smartwatch HR/IBI log, but the selected target is " +
427 sensorType.getDisplayName() + ". Please upload to Smartwatch Heart Rate.")
428 );
429 }
430
431 String sessionCol = findExactColumn(headerMap, "sessionid");
432 String tsCol = findExactColumn(headerMap, "watchtimestampms");
433 String hrCol = findExactColumn(headerMap, "hrbpm");
434 String statusCol = findExactColumn(headerMap, "hrstatus");
435
436 List<String> warnings = new ArrayList<>();
437 int totalRows = 0;
438
439 // Group records by session_id so multi-day cumulative exports isolate the active trip session
440 Map<String, List<SensorDataPoint>> pointsBySession = new LinkedHashMap<>();
441 Map<String, Long> maxEpochBySession = new HashMap<>();
442
443 for (CSVRecord record : parser) {
444 totalRows++;
445 String sessionId = sessionCol != null && record.isMapped(sessionCol) ? record.get(sessionCol).trim() : "default";
446 String rawTs = record.get(tsCol).trim();
447 String rawHr = record.get(hrCol).trim();
448 String rawStatus = statusCol != null && record.isMapped(statusCol) ? record.get(statusCol).trim() : "1";
449
450 if (rawTs.isEmpty() || rawHr.isEmpty()) continue;
451
452 // Only include rows where watch sensor contact status is valid (1) and HR > 0
453 if (!rawStatus.isEmpty() && !rawStatus.equals("1")) {
454 continue;
455 }
456
457 long rawEpochMs;
458 try {
459 rawEpochMs = Long.parseLong(rawTs);
460 } catch (NumberFormatException e) {
461 continue;
462 }
463
464 double hrVal;
465 try {
466 hrVal = Double.parseDouble(rawHr);
467 if (hrVal <= 0 || Double.isNaN(hrVal) || Double.isInfinite(hrVal)) {
468 continue;
469 }
470 } catch (NumberFormatException e) {
471 continue;
472 }
473
474 // Align 1-Hz watch timestamp to integer second boundary (.000) to match 1-Hz GNSS/Polar logs
475 long alignedEpochMs = (rawEpochMs / 1000L) * 1000L;
476 String formatted = DISPLAY_FORMATTER.format(Instant.ofEpochMilli(alignedEpochMs));
477
478 pointsBySession.computeIfAbsent(sessionId, k -> new ArrayList<>())
479 .add(new SensorDataPoint(rawTs, alignedEpochMs, formatted, hrVal));
480 maxEpochBySession.merge(sessionId, alignedEpochMs, Math::max);
481 }
482
483 if (pointsBySession.isEmpty()) {
484 return SensorParseResult.failure(
485 sensorType.getId(),
486 filename,
487 List.of("No valid heart rate measurements (hr_status=1, hr_bpm>0) found in Samsung Watch CSV.")
488 );
489 }
490
491 // Select the latest recording session if multiple historical sessions exist
492 String selectedSession = pointsBySession.keySet().iterator().next();
493 long latestSessionEpoch = Long.MIN_VALUE;
494 for (Map.Entry<String, Long> entry : maxEpochBySession.entrySet()) {
495 if (entry.getValue() > latestSessionEpoch) {
496 latestSessionEpoch = entry.getValue();
497 selectedSession = entry.getKey();
498 }
499 }
500
501 if (pointsBySession.size() > 1) {
502 warnings.add("Multi-session Samsung Watch file detected (" + pointsBySession.size()
503 + " sessions); automatically selected latest trip session (" + selectedSession + ").");
504 }
505
506 // Deduplicate multiple IBI sub-rows sharing the same 1-second timestamp
507 List<SensorDataPoint> rawSessionPoints = pointsBySession.get(selectedSession);
508 Map<Long, SensorDataPoint> uniqueBySecond = new LinkedHashMap<>();
509 for (SensorDataPoint pt : rawSessionPoints) {
510 uniqueBySecond.putIfAbsent(pt.getEpochMillis(), pt);
511 }
512
513 List<SensorDataPoint> points = new ArrayList<>(uniqueBySecond.values());
514 points.sort(Comparator.comparingLong(SensorDataPoint::getEpochMillis));
515
516 double minValue = Double.MAX_VALUE;
517 double maxValue = -Double.MAX_VALUE;
518 for (SensorDataPoint pt : points) {
519 if (pt.getValue() < minValue) minValue = pt.getValue();
520 if (pt.getValue() > maxValue) maxValue = pt.getValue();
521 }
522
523 long minEpoch = points.get(0).getEpochMillis();
524 long maxEpoch = points.get(points.size() - 1).getEpochMillis();
525 String minFormatted = DISPLAY_FORMATTER.format(Instant.ofEpochMilli(minEpoch));
526 String maxFormatted = DISPLAY_FORMATTER.format(Instant.ofEpochMilli(maxEpoch));
527
528 return new SensorParseResult(
529 true,
530 sensorType.getId(),
531 sensorType.getDisplayName(),
532 sensorType.getUnit(),
533 filename,
534 totalRows,
535 points.size(),
536 minValue,
537 maxValue,
538 minEpoch,
539 maxEpoch,
540 minFormatted,
541 maxFormatted,
542 points,
543 List.of(),
544 warnings
545 );
546 }

557 private static SensorParseResult parseHealthyPiGsrLog(
558 CSVParser parser,
559 Map<String, Integer> headerMap,
560 SensorType sensorType,
561 String filename) {
562
563 if (sensorType != SensorType.HEALTHYPI_GSR) {
564 return SensorParseResult.failure(
565 sensorType.getId(),
566 filename,
567 List.of("The uploaded file is a HealthyPi Watch GSR log, but the selected target is " +
568 sensorType.getDisplayName() + ". Please upload to HealthyPi Watch (GSR).")
569 );
570 }
571
572 String tsCol = findExactColumn(headerMap, "timestamputc");
573 String valCol = findExactColumn(headerMap, "ch0detrended");
574 if (valCol == null) {
575 valCol = findExactColumn(headerMap, "ch0");
576 }
577
578 List<SensorDataPoint> allValidPoints = new ArrayList<>();
579 List<SensorDataPoint> exactSecondPoints = new ArrayList<>();
580 Set<Long> seenSeconds = new HashSet<>();
581 List<String> warnings = new ArrayList<>();
582 int totalRows = 0;
583
584 for (CSVRecord record : parser) {
585 totalRows++;
586 if (!record.isMapped(tsCol) || !record.isMapped(valCol)) continue;
587
588 String rawTs = record.get(tsCol).trim();
589 String rawVal = record.get(valCol).trim();
590
591 if (rawTs.isEmpty() || rawVal.isEmpty() || rawTs.equalsIgnoreCase("timestamp_utc")) {
592 continue;
593 }
594
595 ParsedTimestamp parsedTime = parseTimestamp(rawTs);
596 if (parsedTime == null) continue;
597
598 double val;
599 try {
600 val = Double.parseDouble(rawVal);
601 if (Double.isNaN(val) || Double.isInfinite(val)) continue;
602 // Filter out hardware ADC startup saturation spikes (e.g., 536892912 / -221080146)
603 if (Math.abs(val) >= 50000.0) {
604 continue;
605 }
606 } catch (NumberFormatException e) {
607 continue;
608 }
609
610 SensorDataPoint pt = new SensorDataPoint(rawTs, parsedTime.epochMillis, parsedTime.formatted, val);
611 if (totalRows <= 2000) {
612 allValidPoints.add(pt);
613 }
614 if (parsedTime.epochMillis % 1000L == 0 && seenSeconds.add(parsedTime.epochMillis)) {
615 exactSecondPoints.add(pt);
616 }
617 }
618
619 // For 32-Hz multi-thousand-row recordings, use 1-Hz exact-second points (.000Z)
620 List<SensorDataPoint> points = (totalRows > 2000 && !exactSecondPoints.isEmpty())
621 ? exactSecondPoints
622 : allValidPoints;
623
624 if (points.isEmpty()) {
625 return SensorParseResult.failure(
626 sensorType.getId(),
627 filename,
628 List.of("No valid GSR data rows could be parsed from the HealthyPi file.")
629 );
630 }
631
632 points.sort(Comparator.comparingLong(SensorDataPoint::getEpochMillis));
633
634 double minValue = Double.MAX_VALUE;
635 double maxValue = -Double.MAX_VALUE;
636 for (SensorDataPoint pt : points) {
637 if (pt.getValue() < minValue) minValue = pt.getValue();
638 if (pt.getValue() > maxValue) maxValue = pt.getValue();
639 }
640
641 long minEpoch = points.get(0).getEpochMillis();
642 long maxEpoch = points.get(points.size() - 1).getEpochMillis();
643 String minFormatted = DISPLAY_FORMATTER.format(Instant.ofEpochMilli(minEpoch));
644 String maxFormatted = DISPLAY_FORMATTER.format(Instant.ofEpochMilli(maxEpoch));
645
646 return new SensorParseResult(
647 true,
648 sensorType.getId(),
649 sensorType.getDisplayName(),
650 sensorType.getUnit(),
651 filename,
652 totalRows,
653 points.size(),
654 minValue,
655 maxValue,
656 minEpoch,
657 maxEpoch,
658 minFormatted,
659 maxFormatted,
660 points,
661 List.of(),
662 warnings
663 );
664 }

681 public static class ParsedTimestamp {
682 final long epochMillis;
683 final String formatted;
684
685 ParsedTimestamp(long epochMillis, String formatted) {
686 this.epochMillis = epochMillis;
687 this.formatted = formatted;
688 }

691 public static ParsedTimestamp parseTimestampWithZone(String raw, ZoneOffset defaultOffset) {
692 if (raw == null || raw.isEmpty()) return null;
693 for (DateTimeFormatter fmt : DATE_TIME_FORMATTERS) {
694 try {
695 LocalDateTime ldt = LocalDateTime.parse(raw, fmt);
696 Instant instant = ldt.toInstant(defaultOffset);
697 return new ParsedTimestamp(instant.toEpochMilli(), DISPLAY_FORMATTER.format(instant));
698 } catch (Exception ignored) {
699 }
700 }
701 return parseTimestamp(raw);
702 }

704 public static ParsedTimestamp parseTimestamp(String raw) {
705 if (raw == null || raw.isEmpty()) return null;
706
707 // Check if raw is a pure numeric epoch (seconds or milliseconds)
708 if (raw.matches("^-?\\d+(\\.\\d+)?$")) {
709 try {

736 } catch (Exception ignored) {

738
739 // Try Date Time formatters
740 for (DateTimeFormatter fmt : DATE_TIME_FORMATTERS) {
741 try {
742 LocalDateTime ldt = LocalDateTime.parse(raw, fmt);
743 Instant instant = ldt.toInstant(ZoneOffset.UTC);
744 return new ParsedTimestamp(instant.toEpochMilli(), DISPLAY_FORMATTER.format(instant));

746 }
747 }
748
749 // Try LocalTime (if only time like 10:15:30.000 is given)
750 try {
751 LocalTime lt = LocalTime.parse(raw);
752 LocalDate date = LocalDate.of(2026, 10, 11);
753 LocalDateTime ldt = LocalDateTime.of(date, lt);
754 Instant instant = ldt.toInstant(ZoneOffset.UTC);
755 return new ParsedTimestamp(instant.toEpochMilli(), DISPLAY_FORMATTER.format(instant));
756 } catch (Exception ignored) {
