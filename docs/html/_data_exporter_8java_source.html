24 if (result == null || result.getDataPoints() == null || result.getDataPoints().isEmpty()) {

28 StringBuilder sb = new StringBuilder();
29 sb.append("timestamp_iso,formatted_time,epoch_millis,value,sensor_type,unit").append(NEWLINE);
30
31 String sensorId = result.getSensorType() != null ? result.getSensorType() : "";
32 String unit = result.getUnit() != null ? result.getUnit() : "";
33
34 for (SensorDataPoint pt : result.getDataPoints()) {
35 sb.append(escape(pt.getRawTimestamp())).append(CSV_DELIMITER)
36 .append(escape(pt.getFormattedTime())).append(CSV_DELIMITER)
37 .append(pt.getEpochMillis()).append(CSV_DELIMITER)
38 .append(String.format(Locale.US, "%.4f", pt.getValue())).append(CSV_DELIMITER)
39 .append(escape(sensorId)).append(CSV_DELIMITER)
40 .append(escape(unit)).append(NEWLINE);
41 }

50 public static String toMasterCsv(Map<String, SensorParseResult> datasets) {

60 if (datasets == null || datasets.isEmpty()) {

80 for (Map.Entry<Long, String> entry : epochToIso.entrySet()) {
81 long epoch = entry.getKey();

83 String fmt = epochToFmt.getOrDefault(epoch, "");
84
85 Double sw = swMap.get(epoch);

97 .append(acc != null ? String.format(Locale.US, "%.3f", acc) : "").append(CSV_DELIMITER)
98 .append(gsr != null ? String.format(Locale.US, "%.2f", gsr) : "")

171 private static void populateStreamMap(SensorParseResult res,
172 Map<Long, Double> targetMap,
173 Map<Long, String> epochToIso,
174 Map<Long, String> epochToFmt) {
175 if (res == null || res.getDataPoints() == null) return;
176 for (SensorDataPoint pt : res.getDataPoints()) {
177 long ep = pt.getEpochMillis();
178 if (!targetMap.containsKey(ep)) {
179 targetMap.put(ep, pt.getValue());
180 }
181 if (!epochToIso.containsKey(ep) && pt.getRawTimestamp() != null) {
182 epochToIso.put(ep, pt.getRawTimestamp());
183 }
184 if (!epochToFmt.containsKey(ep) && pt.getFormattedTime() != null) {
185 epochToFmt.put(ep, pt.getFormattedTime());
186 }
187 }
188 }

190 private static String sanitizeHeader(String name) {
191 if (name == null) return "value";
192 return name.toLowerCase().replaceAll("[^a-z0-9]+", "_").replaceAll("^_+|_+$", "");
193 }

195 private static String escape(String val) {
196 if (val == null) return "";
197 if (val.contains(",") || val.contains("\"") || val.contains("\n") || val.contains("\r")) {
198 return "\"" + val.replace("\"", "\"\"") + "\"";
199 }
200 return val;
201 }
