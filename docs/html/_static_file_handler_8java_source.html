26 public void handle(HttpExchange exchange) throws IOException {
27 String path = exchange.getRequestURI().getPath();
28 if (path == null || path.equals("/") || path.isEmpty()) {
29 path = "/index.html";
30 }
31
32 // Prevent path traversal
33 if (path.contains("..")) {
34 sendResponse(exchange, 403, "Access Denied", "text/plain");
35 return;
36 }
37
38 byte[] content = null;
39 String contentType = getContentType(path);
40
41 // Try reading from filesystem first (useful for active live edits)
42 Path filePath = filesystemBasePath.resolve(path.startsWith("/") ? path.substring(1) : path);
43 if (Files.exists(filePath) && !Files.isDirectory(filePath)) {
44 content = Files.readAllBytes(filePath);
45 } else {
46 // Fallback to classpath resources
47 String resourcePath = resourceBasePath + (path.startsWith("/") ? path : "/" + path);
48 try (InputStream is = getClass().getResourceAsStream(resourcePath)) {
49 if (is != null) {
50 content = is.readAllBytes();
51 }
52 }
53 }
54
55 if (content == null) {
56 sendResponse(exchange, 404, "404 Not Found: " + path, "text/plain");
57 return;
58 }
59
60 exchange.getResponseHeaders().set("Content-Type", contentType);
61 exchange.getResponseHeaders().set("Cache-Control", "no-cache, no-store, must-revalidate");
62 exchange.sendResponseHeaders(200, content.length);
63 try (OutputStream os = exchange.getResponseBody()) {
64 os.write(content);
65 }
66 }
